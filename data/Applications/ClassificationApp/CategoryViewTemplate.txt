%TMPL:DEF{"pagination"}%<noautolink><div class='clsPagination foswikiHideOnPrint'>%TMPL:P{"pagination::default"
   cats="%TMPL:P{"pagination::cats"
               cat="%CATINFO{
                     cat="%URLPARAM{"catname" default="%BASETOPIC%"}%" 
                     format="$parentname"
                  }%"
            }%"
}%</div></noautolink>%TMPL:END%

%TMPL:DEF{"pagination::cats"}%%HIERARCHY{
   top="%cat%"
   exclude="%cat%"
   depth="2"
   format="$name"
   separator=", "
}%%TMPL:END%

%TMPL:DEF{"pagination::default"}%%TMPL:P{"pagination::prev" 
   cat="%EXTRACT{
            text="%cats%"
            pattern="\b(\w+)\b,\s*%URLPARAM{"catname" default="%BASETOPIC%"}%\b"
            format="$1"
         }%"
}%%TMPL:P{"pagination::separator"}%%TMPL:P{"pagination::next" 
   cat="%EXTRACT{
            text="%cats%"
            pattern="\b%URLPARAM{"catname" default="%BASETOPIC%"}%\b,\s*(\w+)\b"
            format="$1"
         }%"
}%%TMPL:END%

%TMPL:DEF{"pagination::prev"}%%IF{"'%cat%'=''"
   then="<span class='clsPaginationPrev clsPaginationDisabled'></span>"
   else="$percntCATINFO{cat=\"%cat%\" format=\"<a href='$url' class='clsPaginationPrev'>$title</a>\"}$percnt"
}%%TMPL:END%

%TMPL:DEF{"pagination::next"}%%IF{"'%cat%'=''"
   then="<span class='clsPaginationNext clsPaginationDisabled'></span>"
   else="$percntCATINFO{cat=\"%cat%\" format=\"<a href='$url' class='clsPaginationNext'>$title</a>\"}$percnt"
}%%TMPL:END%
