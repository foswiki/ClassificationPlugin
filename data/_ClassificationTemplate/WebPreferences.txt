---++ General settings
   * Set SITEMAPLIST = on
   * Set WEBSUMMARY = 
   * Set NOSEARCHALL = 

---++ Supplemental topics

---++ Application settings
   * Set WEBFORMS = Applications.WikiTopic

---++ Topic creator settings

*TopicType step:*

   * Set TOPICCREATOR_SHOW_DEFAULT_TYPE = off
   * Set TOPICCREATOR_SELECT_TYPE = ClassifiedTopic
   * #Set TOPICCREATOR_EXCLUDE_TYPE = 
   * #Set TOPICCREATOR_INCLUDE_TYPE = 

*<nop>TopicTemplate step:*

   * #Set TOPICCREATOR_SHOW_DEFAULT_TEMPLATE = off
   * #Set TOPICCREATOR_SELECT_TEMPLATE = 
   * #Set TOPICCREATOR_EXCLUDE_TEMPLATE =
   * #Set TOPICCREATOR_INCLUDE_TEMPLATE = 

---++ !ClassificationPlugin settings
   * Set CLASSIFICATIONPLUGIN_AUTOREPARENT = on
   * Set CLASSIFICATIONPLUGIN_RENDERCATEGORY = Applications.ClassificationApp.RenderCategoryTiles
   * Set ENABLEQUALITYCHECK = on
      * possible values: on, tags, cats, h1, href, br

---++ Cache settings
   * Set WEBDEPENDENCIES = WebHome, WebTopicList, WebChanges, WebRss, WebAtom, WebSearch

---++ Access rights
*Web*:
   * Set DENYWEBVIEW = WikiGuest
   * #Set ALLOWWEBVIEW = 
   * Set DENYWEBCHANGE = WikiGuest
   * Set ALLOWWEBCHANGE = AdminGroup
   * #Set DENYWEBRENAME = 
   * Set ALLOWWEBRENAME = AdminGroup
   * Set DENYWEBCOMMENT = WikiGuest

*Topic*
   * Set ALLOWTOPICCHANGE = AdminGroup
   * Set ALLOWTOPICRENAME = AdminGroup
